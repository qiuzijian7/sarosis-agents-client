function rehypeRawSafe() {
  return (tree) => {
    visit(tree, 'raw', (node) => {
      if (typeof DOMParser === 'undefined') return; // Node 环境 no-op
      const doc = new DOMParser().parseFromString(node.value, 'text/html');
      node.type = 'element';
      Object.assign(node, hastFromDom(doc.body, sanitizeChildren));
    });
  };
}

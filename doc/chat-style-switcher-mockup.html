private _storeAgentChatLayout(): void {
  // 保存 group/editor 布局到 workspace storage
  const state: IAgentChatLayoutState = {
    groups: this._serializeGroups(),
    cliMode: this._cliMode,
  };
  this._storageService.store(AGENT_CHAT_LAYOUT_KEY, JSON.stringify(state));
}
